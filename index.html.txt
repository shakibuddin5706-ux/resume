        // Dynamic ADD / DELETE Operations
        function addEducationItem(e) {
            e.preventDefault();
            const newItem = {
                id: Date.now().toString(),
                degree: document.getElementById('newEduDegree').value,
                institution: document.getElementById('newEduInst').value,
                status: document.getElementById('newEduStatus').value,
                description: document.getElementById('newEduDesc').value
            };
            education.push(newItem);
            saveState();
            renderAdminLists();
            e.target.reset();
        }

        function deleteEdu(id) {
            education = education.filter(e => e.id !== id);
            saveState();
            renderAdminLists();
        }

        function addProjectItem(e) {
            e.preventDefault();
            const tagsInput = document.getElementById('newProjTags').value;
            const newItem = {
                id: Date.now().toString(),
                title: document.getElementById('newProjTitle').value,
                category: document.getElementById('newProjCategory').value,
                image: document.getElementById('newProjImg').value || 'https://placehold.co/600x400/0a1128/00f2fe?text=EEE+Project',
                description: document.getElementById('newProjDesc').value,
                tags: tagsInput ? tagsInput.split(',').map(t => t.trim()) : ['EEE', 'Project']
            };
            projects.push(newItem);
            saveState();
            renderAdminLists();
            e.target.reset();
        }

        function deleteProj(id) {
            projects = projects.filter(p => p.id !== id);
            saveState();
            renderAdminLists();
        }

        function addServiceItem(e) {
            e.preventDefault();
            const newItem = {
                id: Date.now().toString(),
                title: document.getElementById('newServiceTitle').value,
                icon: document.getElementById('newServiceIcon').value || 'fa-solid fa-gear',
                desc: document.getElementById('newServiceDesc').value
            };
            services.push(newItem);
            saveState();
            renderAdminLists();
            e.target.reset();
        }

        function deleteService(id) {
            services = services.filter(s => s.id !== id);
            saveState();
            renderAdminLists();
        }

        // View Project Details Modal
        function openProjectModal(id) {
            const p = projects.find(proj => proj.id === id);
            if (!p) return;

            document.getElementById('modalProjTitle').innerText = p.title;
            document.getElementById('modalProjCategory').innerText = p.category;
            document.getElementById('modalProjDesc').innerText = p.description;
            document.getElementById('modalProjImage').src = p.image;

            const tagsContainer = document.getElementById('modalProjTags');
            tagsContainer.innerHTML = (p.tags || []).map(t => `<span class="px-2.5 py-1 rounded bg-slate-800 text-cyan-300 border border-slate-700">${t}</span>`).join('');

            document.getElementById('projectModal').classList.remove('hidden');
        }
